\section*{Nomenclature}

\begin{tabbing}
$A$\qquad\qquad \= Cross-sectional area (m\textsuperscript{2})\\
$b_w$ \> Width of the deck plank cross-section (m)\\
$CIV$ \> Vertical impact coefficient\\
$d$ \> Diameter of the circular girder (m)\\
$esp_{long}$ \> Spacing between girders (m)\\
$esp_{tab}$ \> Spacing between deck planks (m)\\
$E_{0,\mathrm{m}}$ \> Mean modulus of elasticity in bending (kPa)\\
$f_{c0,k}$ \> Characteristic compressive strength parallel to the grain (MPa)\\
$f_{m,d}$ \> Design bending strength (kPa)\\
$f_{v0,d}$ \> Design shear strength parallel to the grain (kPa)\\
$f_1,\ f_2$ \> Objective functions, total timber volume and utilization of the deflection limit\\
$f_m(\mathbf{x})$ \> $m$-th objective function\\
$g_j(\mathbf{x})$ \> $j$-th inequality constraint\\
$h$ \> Height of the deck plank cross-section (m)\\
$I$ \> Second moment of area of the cross-section (m\textsuperscript{4})\\
$k_{mod}$ \> Modification factor\\
$L$ \> Design span of the girder (m)\\
$M_{gk}$ \> Characteristic bending moment due to permanent actions (kN$\cdot$m)\\
$M_{qk}$ \> Characteristic bending moment due to variable actions (kN$\cdot$m)\\
$M_{sd}$ \> Design bending moment (kN$\cdot$m)\\
$P$ \> Wheel load of the design vehicle (kN)\\
$a$ \> Axle spacing of the design vehicle (m)\\
$Q_{sd}$ \> Design shear force (kN)\\
$V_{\mathrm{total}}$ \> Total timber volume of the bridge (m\textsuperscript{3})\\
$W$ \> Section modulus of the cross-section (m\textsuperscript{3})\\
$\mathbf{x}$ \> Vector of design variables\\
$\Lambda,\ \lambda$ \> Deterministic diameter perturbation grid and its generic element\\
$\gamma_g,\ \gamma_q$ \> Partial factors for permanent and variable actions\\
$\gamma_{wf},\ \gamma_{wc}$ \> Material partial factors (bending and shear)\\
$\delta_{\mathrm{tot}}$ \> Total midspan deflection (m)\\
$\delta_{\mathrm{lim}}$ \> Allowable deflection limit (m)\\
$\varphi$ \> Creep coefficient\\
$\psi_2$ \> Quasi-permanent combination factor\\
$\rho$ \> Perturbation factor of the design variable (robustness)\\
$N_c$ \> Number of points of the diameter perturbation grid in the robust evaluation\\
$HV$ \> Hypervolume indicator of the approximated front\\
$S_{T_i}$ \> Total-effect Sobol index of variable $i$
\end{tabbing}
